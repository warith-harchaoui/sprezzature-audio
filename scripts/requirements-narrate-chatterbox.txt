# ChatterboxTTS narration engine — MIT licence on both code and weights.
#
# Install:
#   pip install -r scripts/requirements-narrate-chatterbox.txt
#   sprezzature-audio-install-voice --engine chatterbox
#
# Notes:
# - ChatterboxTTS is published on PyPI as ``chatterbox-tts``. The
#   import path is ``chatterbox.tts`` (or ``chatterbox_tts`` on older
#   releases — the wrapper tries both).
# - torch + torchaudio are pulled in transitively. CPU is fine for
#   narration use; install a GPU build separately for faster inference.
# - First synthesis call downloads ~1 GB of weights into
#   ~/.cache/huggingface/ via the HuggingFace hub client.

chatterbox-tts>=0.1.0
torch>=2.0.0
torchaudio>=2.0.0
numpy>=1.24.0

# For sprezzature-publish's narrate_post.py, which drives these engines and
# routes its optional LLM enrichment of the narration hints through
# best_engine_ai_helper.llm.chat at module level, so it cannot be imported
# without it. Nothing in this package imports it: the engine wrappers here
# only synthesise. Kept so that installing an engine tier gives a working
# narration chain end to end rather than one that fails at the caller.
best-engine-ai-helper>=1.4
